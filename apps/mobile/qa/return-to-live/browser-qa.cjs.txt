// Browser QA for the floating Return to live button. Rename to .cjs and run from the repository root:
//   node apps/mobile/qa/design-system/mock-reader-server.cjs &
//   CI=1 EXPO_PUBLIC_API_URL=https://reader.invalid EXPO_PUBLIC_ACTIONCABLE_URL=wss://reader.invalid/cable \
//     npx --prefix apps/mobile expo export --platform web --clear --output-dir <export>
//   serve the base-commit export on 127.0.0.1:8096 and this branch's on 127.0.0.1:8095 with single-page fallback,
//   then: node browser-qa.cjs <output dir>
// API requests to https://reader.invalid are proxied to the synthetic loopback fixture on 127.0.0.1:8100.
// No production host, account, or worship service is contacted.
const { chromium } = require(process.env.PLAYWRIGHT_MODULE || 'playwright')
const out = process.argv[2]
;(async () => {
  const browser = await chromium.launch()
  for (const [label, port] of [['before', 8096], ['after', 8095]]) {
    for (const [shape, viewport] of [['portrait', { width: 390, height: 844 }], ['landscape', { width: 844, height: 390 }]]) {
      const page = await browser.newPage({ viewport, deviceScaleFactor: 2, hasTouch: false })
      const errors = []; page.on('pageerror', e => errors.push(e.message))
      await page.route('https://reader.invalid/**', async route => {
        const url = new URL(route.request().url())
        await route.fulfill({ response: await route.fetch({ url: 'http://127.0.0.1:8100' + url.pathname + url.search }) })
      })
      await page.routeWebSocket('wss://reader.invalid/cable', ws => {
        ws.send(JSON.stringify({ type: 'welcome' }))
        ws.onMessage(data => { const f = JSON.parse(data); if (f.command === 'subscribe') ws.send(JSON.stringify({ identifier: f.identifier, type: 'confirm_subscription' })) })
      })
      await page.goto(`http://127.0.0.1:${port}/sessions/9001/parts`)
      await page.waitForTimeout(3000)
      await page.getByRole('button', { name: 'Not now' }).click({ timeout: 3000 }).catch(() => {})
      await page.waitForTimeout(500)
      await page.mouse.move(viewport.width / 2, viewport.height / 2)
      for (let i = 0; i < 4; i++) { await page.mouse.wheel(0, 250); await page.waitForTimeout(150) }
      await page.getByRole('button', { name: 'Return to live' }).waitFor({ timeout: 8000 }).catch(e => errors.push('no button: ' + e.message.split('\n')[0]))
      await page.waitForTimeout(1500)
      await page.screenshot({ path: `${out}/${label}-${shape}.png` })
      console.log(label, shape, errors.join(' | ') || 'ok')
      await page.close()
    }
  }
  await browser.close()
})()
