const {chromium}=require(process.env.PLAYWRIGHT_MODULE || 'playwright');const fs=require('fs');const assert=require('assert/strict');
(async()=>{const browser=await chromium.launch({headless:true,channel:'chrome'});try{const page=await browser.newPage({viewport:{width:390,height:844}});const errors=[];page.on('pageerror',e=>errors.push(e.message));await page.route('**/*.ttf',r=>r.abort());await page.goto('http://localhost:8087');await page.getByRole('button',{name:'Retry reading font'}).waitFor();assert(await page.getByText('A space for prayer.',{exact:true}).isVisible());const armenian=page.locator('[lang="hy"]').first();assert(!(await armenian.evaluate(e=>getComputedStyle(e).fontFamily)).includes('Noto'));await page.screenshot({path:process.cwd()+'/apps/mobile/qa/expo73/font-fallback.png'});await page.getByRole('textbox',{name:'Reading note',exact:true}).fill('Keep me through retry');await page.unroute('**/*.ttf');await page.getByRole('button',{name:'Retry reading font'}).click();await page.getByRole('button',{name:'Retry reading font'}).waitFor({state:'hidden'});await page.waitForFunction(()=>[...document.querySelectorAll('[lang="hy"]')].some(e=>getComputedStyle(e).fontFamily.includes('Noto')));assert.equal(await page.getByRole('textbox',{name:'Reading note',exact:true}).inputValue(),'Keep me through retry');assert.deepEqual(errors,[]);fs.writeFileSync(process.cwd()+'/apps/mobile/qa/expo73/font-fallback.json',JSON.stringify({result:'passed',injectedFailure:'Aborted both local TTF requests; expected network console errors only',assertions:['Gallery remains mounted and interactive','System font selected with Armenian caveat and accessible retry','Retry loads Noto without losing typed note'],uncaughtErrors:errors},null,2));console.log('Fallback and retry passed')}finally{await browser.close()}})().catch(e=>{console.error(e);process.exit(1)});
