import React, { useEffect, useState, useRef, useCallback } from 'react';
import api from '../api';
import usePusher from '../hooks/usePusher';
import Navbar from '../components/Navbar';
import { fromJSON } from 'postcss';


const PartsList = ({ sessionId }) => {
  const [parts, setParts] = useState([]);
  const [selectedLang, setSelectedLang] = useState('en'); // Default language
  const partsRef = useRef({});

  const fetchParts = useCallback(async () => {
    try {
      const response = await api.get(`/s/${sessionId}/parts.json`);
      setParts(response.data.parts);
      const sessionResponse = await api.get(`/s/${sessionId}.json`);
      handlePartUpdated({ part: sessionResponse.data.session.current_part_id });
    } catch (error) {
      console.error('Error fetching parts:', error);
    }
  }, [sessionId]);

  useEffect(() => {
    fetchParts();
  }, [fetchParts]);

  let focusedElement = null;

  const handlePartUpdated = (data) => {
    const parsedData = typeof data === 'string' ? JSON.parse(data) : data;
    const { session, liturgy, part } = parsedData;
    const element = partsRef.current[parsedData.part];
    if (element) {
      const stickyElement = document.querySelector('.sticky');
      const offset = stickyElement.offsetHeight;
      console.log('offset', offset);


      // Remove the focus class from the previously focused element
      if (focusedElement) {
        focusedElement.classList.remove('mt-14', 'focus');
      }

      // Add the focus class to the new element
      element.classList.add('focus', 'mt-14', 'animate-flash');

      // Store a reference to the new focused element
      focusedElement = element;

      setTimeout(() => {
        element.classList.remove('animate-flash');
      }, 1000); // Duration of the animation

      element.scrollIntoView({ behavior: 'auto', block: 'start' });
      requestAnimationFrame(() => {
        window.scrollBy(0, -offset);
      });
    }
  };

  usePusher(`channel_${sessionId}`, 'part_change', handlePartUpdated);

  const renderPartContent = (part) => {
    switch (selectedLang) {
      case 'grabar':
        return <p>{part.grabar}</p>;
      case 'translit':
        return <p>{part.translit}</p>;
      case 'en':
        return <p>{part.en}</p>;
      case 'ru':
        return <p>{part.ru}</p>;
      default:
        return <p>{part.en}</p>;
    }
  };

  return (
    <div>
      <Navbar />
      <div className="bg-gray-800 text-white p-4 rounded-lg shadow-md">
      <div className="sticky top-0 flex justify-center mb-6 bg-gray-800 py-2 z-10  rounded">
        <button className={`px-4 py-2 rounded-l ${selectedLang === 'grabar' ? 'bg-gray-700' : 'bg-gray-600'}`} onClick={() => setSelectedLang('grabar')}>Գրաբար</button>
        <button className={`px-4 py-2 ${selectedLang === 'translit' ? 'bg-gray-700' : 'bg-gray-600'}`} onClick={() => setSelectedLang('translit')}>Translit</button>
        <button className={`px-4 py-2 ${selectedLang === 'en' ? 'bg-gray-700' : 'bg-gray-600'}`} onClick={() => setSelectedLang('en')}>English</button>
        <button className={`px-4 py-2 rounded-r ${selectedLang === 'ru' ? 'bg-gray-700' : 'bg-gray-600'}`} onClick={() => setSelectedLang('ru')}>Русский</button>
      </div>
        <ul className="space-y-4">
          {parts.map(({ part }) => (
            <li key={part.id} ref={(el) => (partsRef.current[part.id] = el)} className="bg-gray-700 p-4 rounded shadow hover:bg-gray-600 transition">
              <h2 className="font-semibold capitalize">{part.role}</h2>
              {renderPartContent(part)}
            </li>
          ))}
        </ul>
      </div>
    </div>
  );
};

export default PartsList;

      </label>

        Password:

      </label>
      <button type="submit" className="bg-blue-500 text-white p-2 rounded">Sign Up</button>
    </form>
  );
};

export default SignUp;

import React, { useEffect, useState } from 'react';
import { Link } from 'react-router-dom';
import api from '../api';

const SessionList = () => {
  const [sessions, setSessions] = useState([]);

  useEffect(() => {
    const fetchSessions = async () => {
      try {
        const response = await api.get('/services.json');
        // console.log(response.data);
        setSessions(response.data);
      } catch (error) {
        console.error('Error fetching sessions:', error);
      }
    };

    fetchSessions();
  }, []);

  return (
    <div>
      <h1 className="text-2xl font-bold mb-4">Current Services</h1>
      <ul className="space-y-4">
        {Array.isArray(sessions) && sessions.length > 0 ? (
          sessions.map(({ session }) => (
            <li key={session.id} className="border p-4 rounded shadow hover:bg-gray-100 transition bg-gray-800 shadow-md">
              <Link to={`/sessions/${session.id}/parts`} className="flex items-center">
                <img src={session.avatar_url} alt="Avatar" className="w-16 h-16 rounded-full mr-4" />
                <div>
                  <h2 className="text-xl font-semibold text-blue-500 hover:underline">{session.liturgy}</h2>
                  <p className="text-gray-500 text-sm">{session.date} - {session.church_name}</p>
                </div>
              </Link>
            </li>
          ))
        ) : (
          <li className='border p-4 rounded shadow hover:bg-gray-100 transition bg-gray-800 shadow-md'>
            <p>No services are currently taking place.</p>
          </li>
        )}
      </ul>
    </div>
  );
};

export default SessionList;

import React, { useEffect, useState } from 'react';
import { useParams } from 'react-router-dom';
import api from '../api';
import usePusher from '../hooks/usePusher';
import ReactMarkdown from 'react-markdown';


const Explanations = () => {
  const [explanation, setExplanation] = useState(null);

  // Get the session ID from the URL
  const { id: sessionId } = useParams();
  const [logData, setLogData] = useState('');


  const fetchExplanation = async () => {
    try {
      const response = await api.get(`/s/${sessionId}/explanations/current.json`);
      setExplanation(response.data[0]);
    } catch (error) {
      console.error('Error fetching explanation:', error);
    }
  };

  useEffect(() => {
    fetchExplanation();
  }, [sessionId]);

  usePusher(`channel_${sessionId}`, 'part_change', () => {
    fetchExplanation();
  })

  if (!explanation) {
    return (
      <div>
        Loading... 
      </div>
    );
  }

  return (
    <div className="bg-gray-100 text-gray-600 p-4 rounded-lg shadow-md w-full">
      <h2 className="text-2xl font-bold mb-2">{explanation.title}</h2>
      <ReactMarkdown>{explanation.description}</ReactMarkdown>
    </div>
  );
};

export default Explanations;

FILE: ./src/hooks/usePusher.js


import { useEffect } from 'react';
import pusher from '../pusher';

const usePusher = (channelName, eventName, callback) => {
  useEffect(() => {
    const channel = pusher.subscribe(channelName);

    channel.bind(eventName, callback);

    return () => {
      channel.unbind(eventName, callback);
      pusher.unsubscribe(channelName);
    };
  }, [channelName, eventName, callback]);
};

export default usePusher;
